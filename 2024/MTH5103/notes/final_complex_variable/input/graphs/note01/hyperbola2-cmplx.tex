\documentclass{standalone}
\usepackage{tikz}
\usepackage{amsmath}

\begin{document}
\begin{tikzpicture}[scale=1]

  % Axes
  \draw[->] (-4,0) -- (4,0);
  \draw[->] (0,-4) -- (0,4);

  % Define 'a' and 'b'
  \def\a{1.5}
  \def\b{2}

  % Asymptotes (y = ±(a/b)x)
  \draw[blue!70] (-4,-4*\a/\b) -- (4,4*\a/\b);
  \draw[blue!70] (-4,4*\a/\b) -- (4,-4*\a/\b);

  % Hyperbola (longer branches, adjusted domain)
  \begin{scope}
    \clip (-4,-4) rectangle (4,4);
    \draw[red, thick, domain=\b:3.5, samples=100] % Increased upper bound of domain
      plot ({\a*sqrt((\x*\x/\b/\b) - 1)}, \x);
    \draw[red, thick, domain=\b:3.5, samples=100] % Increased upper bound of domain
      plot ({-\a*sqrt((\x*\x/\b/\b) - 1)}, \x);
    \draw[red, thick, domain=-\b:-3.5, samples=100] % Decreased lower bound of domain
      plot ({\a*sqrt((\x*\x/\b/\b) - 1)}, \x);
    \draw[red, thick, domain=-\b:-3.5, samples=100] % Decreased lower bound of domain
      plot ({-\a*sqrt((\x*\x/\b/\b) - 1)}, \x);
  \end{scope}

  % Points of intersection with y-axis (0, b) and (0, -b)
  \draw[fill=black] (0,\b) circle (1pt) node[right] {$b$};
  \draw[fill=black] (0,-\b) circle (1pt) node[right] {$-b$};

\end{tikzpicture}
\end{document}
